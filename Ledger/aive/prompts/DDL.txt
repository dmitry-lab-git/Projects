accounts	CREATE TABLE dicts.accounts (code integer, account_name character varying(50), account_id character varying(50), entity_id character varying(50), id uuid NOT NULL, opening_date date, keywords text, created_at timestamp with time zone, status_id uuid NOT NULL);
addresses	CREATE TABLE dicts.addresses (code integer, adress_name character varying(64), address_id character varying(50), external_status character varying(50), observed_at character varying(50), city character varying(50), country character varying(50), id uuid NOT NULL, created_at timestamp with time zone, status_id uuid NOT NULL);
contragents	CREATE TABLE dicts.contragents (code integer NOT NULL, contragent_name character varying(50), contragent_id character varying(50), contragent_type character varying(50), description character varying(512), adress_id character varying(50), id uuid NOT NULL, address_id uuid, keywords ARRAY, created_at timestamp with time zone, status_id uuid NOT NULL);
dict_categories	CREATE TABLE dicts.dict_categories (code integer NOT NULL, category_name character varying(255) NOT NULL, category_id character varying(50) NOT NULL, parent_category_id character varying(50), keywords ARRAY, created_at timestamp with time zone, id uuid NOT NULL, status_id uuid NOT NULL);
dict_ledger_versions	CREATE TABLE dicts.dict_ledger_versions (code integer NOT NULL, version_name text NOT NULL, version_code text NOT NULL, registration_date date NOT NULL, description text, created_at timestamp with time zone, id uuid NOT NULL, status_id uuid NOT NULL);
dict_types	CREATE TABLE dicts.dict_types (code integer, type_name character varying(50), type_id character varying(50), type_code character varying(50), net_value_effect character varying(50), account_effect numeric, needs_entity character varying(50), needs_project character varying(50), created_at timestamp with time zone, id uuid NOT NULL, status_id uuid NOT NULL);
entities	CREATE TABLE dicts.entities (code integer, entity_name character varying(50), entity_id character varying(50), category_id character varying(50), keywords ARRAY, created_at timestamp with time zone, id uuid NOT NULL, status_id uuid NOT NULL);
responsibles	CREATE TABLE dicts.responsibles (code integer, resp_name character varying(50), responsible_id character varying(50), created_at timestamp with time zone, id uuid NOT NULL, status_id uuid NOT NULL);
