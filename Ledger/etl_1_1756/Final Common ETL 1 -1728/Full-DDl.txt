-- ====================================================
-- TABLE: public.acc_balance_snapshot_register
-- ====================================================

CREATE TABLE public.acc_balance_snapshot_register (
    code                           integer NOT NULL,
    account_id                     character varying(50),
    tr_id                          character varying(50),
    balance                        integer,
    registration_date              date,
    legacy_code                    integer,
    created_at                     timestamp with time zone,
    id                             uuid
);

ALTER TABLE public.acc_balance_snapshot_register ADD PRIMARY KEY (code);

-- ====================================================
-- TABLE: public.ledger
-- ====================================================

CREATE TABLE public.ledger (
    transaction_id                 integer NOT NULL,
    transaction_uuid               uuid NOT NULL,
    name                           character varying(512),
    operation_date                 date NOT NULL,
    category_id                    character varying(50),
    amount                         numeric(19,4) NOT NULL,
    type_id                        character varying(50),
    contragent_from_id             character varying(50),
    contragent_to_id               character varying(50),
    account_id                     character varying(50),
    acc_balance                    numeric(19,4) NOT NULL,
    system_balance                 numeric(19,4) NOT NULL,
    prev_acc_tr_id                 integer,
    prev_acc_tr_balance            numeric(19,4),
    check_sum                      character varying(64) NOT NULL
);

ALTER TABLE public.ledger ADD PRIMARY KEY (transaction_id);

CREATE INDEX idx_ledger_acc_date ON public.ledger USING btree (account_id, operation_date);
CREATE INDEX idx_ledger_tr_uuid ON public.ledger USING btree (transaction_uuid);

-- ====================================================
-- TABLE: public.transactions
-- ====================================================

CREATE TABLE public.transactions (
    id                             uuid NOT NULL,
    code                           integer NOT NULL,
    parent_id                      uuid,
    transaction_name               character varying(1024),
    amount                         numeric(15,2) NOT NULL,
    currency_id                    uuid NOT NULL,
    operation_date                 date NOT NULL,
    operation_time                 time without time zone,
    registration_date              date NOT NULL,
    type_id                        uuid NOT NULL,
    entity_id                      uuid NOT NULL,
    category_id                    uuid NOT NULL,
    responsible_id                 uuid NOT NULL,
    contragent_from_id             uuid,
    contragent_to_id               uuid,
    account_id                     uuid NOT NULL,
    source_extraction_id           uuid NOT NULL,
    tx_hash                        character varying(64),
    status                         character varying(50) NOT NULL,
    created_at                     timestamp without time zone NOT NULL
);

ALTER TABLE public.transactions ADD UNIQUE (code);
ALTER TABLE public.transactions ADD FOREIGN KEY (parent_id) REFERENCES transactions(id);
ALTER TABLE public.transactions ADD FOREIGN KEY (currency_id) REFERENCES dicts.dicts_currencies(id);
ALTER TABLE public.transactions ADD FOREIGN KEY (type_id) REFERENCES dicts.dict_types(id);
ALTER TABLE public.transactions ADD FOREIGN KEY (entity_id) REFERENCES dicts.entities(id);
ALTER TABLE public.transactions ADD FOREIGN KEY (category_id) REFERENCES dicts.dict_categories(id);
ALTER TABLE public.transactions ADD FOREIGN KEY (responsible_id) REFERENCES dicts.responsibles(id);
ALTER TABLE public.transactions ADD FOREIGN KEY (contragent_from_id) REFERENCES dicts.contragents(id);
ALTER TABLE public.transactions ADD FOREIGN KEY (contragent_to_id) REFERENCES dicts.contragents(id);
ALTER TABLE public.transactions ADD FOREIGN KEY (account_id) REFERENCES dicts.accounts(id);
ALTER TABLE public.transactions ADD PRIMARY KEY (id);

CREATE UNIQUE INDEX transactions_code_key ON public.transactions USING btree (code);
